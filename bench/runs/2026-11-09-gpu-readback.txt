# 2. What drives the collections: keiei (Apple M2, apple metal-3), node v26.8.2, load 5.7-7.6, at 746a9124
#    with this uncommitted change, so each run's values come out of the mapped range one of three ways:
#      `copy` as shipped (a fresh ArrayBuffer per channel per frame: 32 MB a frame at 1M), `view` (no copy:
#      a view onto the mapping, emptied by the unmap; the bench never reads values), and `into` (copied
#      into one Float32Array per channel, made once and reused every frame).
#    --- src/gpu/readback.ts, in `Readback.copy`'s `end`:
#    -            const values = new Float32Array(range.slice(p.at, p.at + p.bytes));
#    +            const mode = (globalThis as { blitsReadMode?: string }).blitsReadMode;
#    +            const pool = ((globalThis as { blitsReadPool?: Map<string, Float32Array> }).blitsReadPool ??= new Map());
#    +            let values: Float32Array;
#    +            if (mode === 'view') values = new Float32Array(range, p.at, p.bytes / 4);
#    +            else if (mode === 'into') {
#    +              let into = pool.get(p.name);
#    +              if (into === undefined || into.length !== p.bytes / 4) pool.set(p.name, (into = new Float32Array(p.bytes / 4)));
#    +              into.set(new Float32Array(range, p.at, p.bytes / 4));
#    +              values = into;

# Measured: the spikes are major collections the reads' fresh arrays bring on. Copying the same 32 MB a
# frame into arrays that are reused leaves no frame over 16.7 ms and takes 2.0-2.2 ms off the median.
# `GpuRead.values` is promised never to be written again, so reusing one breaks that promise; a fix
# needs a way for the host to hand back, or hand in, the array a read fills.
